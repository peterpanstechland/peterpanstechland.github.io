"""Pack 1-4 local photos into a private RGB565 Indicator firmware header.

Images are oriented, center-cropped to 400x280, and stripped of metadata.
Requires Pillow. Never uploads images. Rebuild/flash the ESP32 application after
packing; this is not a runtime phone-to-screen transfer service.
"""
from __future__ import annotations

import argparse
import json
from pathlib import Path
import unicodedata

from PIL import Image, ImageOps

ROOT = Path(__file__).resolve().parents[1]
WIDTH, HEIGHT, MAX_PHOTOS = 400, 280, 4
DEFAULT_OUTPUT = ROOT / "firmware/indicator/include/photos.local.h"


def caption_for(path: Path) -> str:
    # The built-in font covers BMP text; unsupported emoji is not emitted.
    value = "".join(c for c in path.stem if c.isprintable() and ord(c) <= 0xFFFF
                    and not unicodedata.category(c).startswith("C"))
    return value[:24] or "照片"


def read_pixels(path: Path) -> list[int]:
    with Image.open(path) as source:
        upright = ImageOps.exif_transpose(source)
        rgba = upright.convert("RGBA")
        matte = Image.new("RGBA", rgba.size, (244, 241, 222, 255))
        matte.alpha_composite(rgba)
        fitted = ImageOps.fit(matte.convert("RGB"), (WIDTH, HEIGHT),
                              method=Image.Resampling.LANCZOS)
        # Numerical RGB565 words; the C++ driver handles native word order.
        # tobytes works across supported Pillow versions without the deprecated
        # getdata API; no byte-order-dependent uint16 conversion is used.
        raw = fitted.tobytes()
        return [((raw[i] >> 3) << 11) | ((raw[i+1] >> 2) << 5) | (raw[i+2] >> 3)
                for i in range(0, len(raw), 3)]


def pack(paths: list[Path], output: Path) -> int:
    if not 1 <= len(paths) <= MAX_PHOTOS:
        raise ValueError(f"Choose 1-{MAX_PHOTOS} photos")
    # Decode every input before writing: a bad file leaves the old pack intact.
    photos = [(caption_for(path), read_pixels(path)) for path in paths]
    lines = ["// Private photo pack; generated by tools/pack_indicator_photos.py.",
             "// RGB565 pixels only. No EXIF, camera, location, or source path.",
             "#pragma once", "#include <Arduino.h>", "namespace local_photos {",
             f"constexpr int16_t kWidth = {WIDTH}, kHeight = {HEIGHT};",
             "struct Photo { const uint16_t *pixels; const char *caption; };"]
    for i, (_, pixels) in enumerate(photos):
        lines.append(f"static const uint16_t photo{i}[] PROGMEM = {{")
        for pos in range(0, len(pixels), 16):
            lines.append("  " + ",".join(f"0x{pixel:04X}" for pixel in pixels[pos:pos+16]) + ",")
        lines.append("};")
    lines.append("static const Photo kPhotos[] = {")
    for i, (caption, _) in enumerate(photos):
        lines.append(f"  {{photo{i}, {json.dumps(caption, ensure_ascii=False)}}},")
    lines += ["};", "constexpr size_t kCount = sizeof(kPhotos) / sizeof(kPhotos[0]);", "}", ""]
    output.parent.mkdir(parents=True, exist_ok=True)
    temporary = output.with_suffix(output.suffix + ".tmp")
    temporary.write_text("\n".join(lines), encoding="utf-8")
    temporary.replace(output)
    return len(photos) * WIDTH * HEIGHT * 2


def main() -> None:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("photos", nargs="+", type=Path)
    parser.add_argument("--output", type=Path, default=DEFAULT_OUTPUT)
    args = parser.parse_args()
    try:
        size = pack(args.photos, args.output)
    except (OSError, ValueError) as error:
        parser.error(str(error))
    print(f"Packed {len(args.photos)} photos, {size:,} pixel bytes: {args.output}")
    print("Run tools/build_indicator_font.py for caption glyphs, then rebuild Indicator.")


if __name__ == "__main__":
    main()
