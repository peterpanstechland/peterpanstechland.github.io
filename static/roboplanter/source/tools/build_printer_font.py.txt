"""Generate a 32 px Noto Sans SC receipt font: printable ASCII plus GB2312.

Firmware sends pixels, not GBK text; printer-resident Chinese fonts are unused.
Pillow and the source OTF are needed only when regenerating this header.
"""
from __future__ import annotations
import argparse
import hashlib
import math
from pathlib import Path
from PIL import Image, ImageDraw, ImageFont

ROOT = Path(__file__).resolve().parents[1]
DEFAULT_FONT = ROOT / ".tools/fonts/NotoSansSC-Regular.otf"
DEFAULT_OUTPUT = ROOT / "firmware/xiao/include/printer_font.h"
PIXEL_SIZE, BASELINE, LINE_HEIGHT, THRESHOLD = 32, 32, 44, 100

def characters() -> list[str]:
    chars = set(chr(c) for c in range(32, 127))
    for high in range(0xA1, 0xF8):
        for low in range(0xA1, 0xFF):
            try:
                char = bytes((high, low)).decode("gb2312")
            except UnicodeDecodeError:
                continue
            if char.isprintable() or char == "\u3000":
                chars.add(char)
    return sorted(chars, key=ord)

def rasterize(font: ImageFont.FreeTypeFont, char: str) -> dict:
    left, top, right, bottom = font.getbbox(char, anchor="ls")
    canvas = Image.new("L", (max(1, right-left)+4, max(1, bottom-top)+4))
    ImageDraw.Draw(canvas).text((2-left, 2-top), char, font=font, fill=255, anchor="ls")
    mask = canvas.point(lambda value: 255 if value >= THRESHOLD else 0)
    advance = max(1, math.ceil(font.getlength(char)))
    bbox = mask.getbbox()
    if bbox is None:
        if char not in (" ", "\u3000"):
            raise ValueError(f"Empty glyph U+{ord(char):04X}")
        return dict(advance=advance, width=0, height=0, x=0, y=0, bitmap=b"")
    ink = mask.crop(bbox)
    x, y = left-2+bbox[0], top-2+bbox[1]+BASELINE
    width, height = ink.size
    if not (0 <= y and y+height <= LINE_HEIGHT and -8 <= x
            and x+width <= advance+8 and advance <= 48):
        raise ValueError(f"Glyph outside receipt cell U+{ord(char):04X}")
    stride = (width+7)//8
    packed = bytearray(stride*height)
    for row in range(height):
        for col in range(width):
            if ink.getpixel((col,row)):
                packed[row*stride+col//8] |= 0x80 >> (col%8)
    return dict(advance=advance, width=width, height=height, x=x, y=y, bitmap=bytes(packed))

def generate(font_path: Path, output: Path) -> tuple[int, int]:
    font = ImageFont.truetype(str(font_path), PIXEL_SIZE)
    missing = rasterize(font, "\uffff")
    chars, packed, entries = characters(), bytearray(), []
    for char in chars:
        g = rasterize(font, char)
        if g == missing:
            raise ValueError(f"Font lacks U+{ord(char):04X}")
        entries.append("  {0x%04X, %d, %d, %d, %d, %d, %d}," %
                       (ord(char), g["advance"], g["width"], g["height"], g["x"], g["y"], len(packed)))
        packed.extend(g["bitmap"])
    digest = hashlib.sha256(font_path.read_bytes()).hexdigest()
    lines = ["// Generated by tools/build_printer_font.py; do not edit.",
             "// Noto Sans SC, SIL OFL 1.1: ../../indicator/assets/fonts/OFL.txt",
             f"// Source font SHA256: {digest}",
             "// Printable ASCII + GB2312; 32 px monochrome, MSB-first rows.",
             "#pragma once", "#include <cstddef>", "#include <cstdint>",
             "namespace printer_font {",
             "struct Glyph { uint16_t code; uint8_t advance, width, height; int8_t x, y; uint32_t offset; };",
             "static const Glyph kGlyphs[] = {"]
    lines.extend(entries)
    lines += ["};", "constexpr size_t kCount = sizeof(kGlyphs)/sizeof(kGlyphs[0]);",
              f"constexpr size_t kBitmapBytes = {len(packed)};",
              f"constexpr uint8_t kLineHeight = {LINE_HEIGHT};", "static const uint8_t kBitmap[] = {"]
    lines += ["  "+",".join(f"0x{b:02X}" for b in packed[i:i+24])+"," for i in range(0,len(packed),24)]
    lines += ["};", "static_assert(sizeof(kBitmap) == kBitmapBytes, \"Incomplete printer bitmap\");",
              "inline const Glyph* find(uint16_t code) {", "  size_t lo = 0, hi = kCount;",
              "  while (lo < hi) {", "    const size_t mid = lo + (hi-lo)/2;",
              "    if (kGlyphs[mid].code < code) lo = mid+1; else hi = mid;", "  }",
              "  return lo < kCount && kGlyphs[lo].code == code ? &kGlyphs[lo] : nullptr;", "}", "}", ""]
    output.parent.mkdir(parents=True, exist_ok=True)
    output.write_text("\n".join(lines), encoding="utf-8")
    return len(chars), len(packed)

if __name__ == "__main__":
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--font", type=Path, default=DEFAULT_FONT)
    parser.add_argument("--output", type=Path, default=DEFAULT_OUTPUT)
    args = parser.parse_args()
    count, size = generate(args.font, args.output)
    print(f"Generated {count} receipt glyphs, {size} bitmap bytes")
