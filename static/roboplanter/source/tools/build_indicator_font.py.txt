"""Generate native-size, cropped monochrome fonts used by Indicator.

Requires Pillow. The generated header is shipped with the firmware, so normal
PlatformIO builds do not require Pillow or the source OTF.
"""
from __future__ import annotations

import argparse
import hashlib
import math
from pathlib import Path
import re

from PIL import Image, ImageDraw, ImageFont

ROOT = Path(__file__).resolve().parents[1]
DEFAULT_FONT = ROOT / ".tools/fonts/NotoSansSC-Regular.otf"
DEFAULT_OUTPUT = ROOT / "firmware/indicator/include/ui_glyphs.h"
PIXEL_SIZES = (16, 24, 32, 48, 64)
THRESHOLD = 100


def required_characters() -> list[str]:
    chars = set(chr(c) for c in range(32, 127))
    for folder in (ROOT / "firmware/indicator/src", ROOT / "firmware/indicator/include"):
        for source in sorted(folder.glob("*")):
            if source.suffix not in (".cpp", ".h") or source.name == "ui_glyphs.h":
                continue
            # Include source comments too: small conservative over-inclusion is
            # preferable to silently missing a visible label or local caption.
            chars.update(c for c in source.read_text(encoding="utf-8")
                         if ord(c) > 127 and c.isprintable() and ord(c) <= 0xFFFF)
    return sorted(chars, key=ord)


def legacy_advance(font: ImageFont.FreeTypeFont, char: str) -> int:
    """Exactly the original 16 px layout, including its ASCII width cap."""
    return 16 if ord(char) >= 0x2E80 else min(16, max(1, math.ceil(font.getlength(char))))


def rasterize(font: ImageFont.FreeTypeFont, char: str) -> dict:
    """Rasterize at the requested size, keeping all ink and its true bearings.

    A glyph's nominal top-left remains baseline 14/16 em. Its ink may extend
    left of that origin or below one em; cropping never changes that position.
    Rows are byte-aligned, MSB first, and contain thresholded native pixels.
    """
    size = font.size
    left, top, right, bottom = font.getbbox(char, anchor="ls")
    # Extra margin makes the raster independent of a nominal square cell.
    margin = 2
    canvas = Image.new("L", (max(1, right - left) + 2 * margin,
                             max(1, bottom - top) + 2 * margin))
    ImageDraw.Draw(canvas).text((margin - left, margin - top), char,
                               font=font, fill=255, anchor="ls")
    mask = canvas.point(lambda value: 255 if value >= THRESHOLD else 0)
    bbox = mask.getbbox()
    advance = size if ord(char) >= 0x2E80 else max(1, math.ceil(font.getlength(char)))
    if bbox is None:
        if not char.isspace():
            raise ValueError(f"Empty glyph: U+{ord(char):04X} at {size} px")
        return dict(advance=advance, width=0, height=0, x=0, y=0, bitmap=b"")
    ink = mask.crop(bbox)
    x_offset = left - margin + bbox[0]
    y_offset = top - margin + bbox[1] + (14 * size // 16)
    width, height = ink.size
    if not (0 <= width <= 255 and 0 <= height <= 255 and 0 <= advance <= 255
            and -128 <= x_offset <= 127 and -128 <= y_offset <= 127):
        raise ValueError(f"Glyph metadata overflow: U+{ord(char):04X} at {size} px")
    stride = (width + 7) // 8
    data = bytearray(stride * height)
    for y in range(height):
        for x in range(width):
            if ink.getpixel((x, y)):
                data[y * stride + x // 8] |= 0x80 >> (x % 8)
    return dict(advance=advance, width=width, height=height,
                x=x_offset, y=y_offset, bitmap=bytes(data))


def generate(font_path: Path, output: Path, preview: Path | None = None) -> int:
    legacy_font = ImageFont.truetype(str(font_path), 16)
    chars = required_characters()
    digest = hashlib.sha256(font_path.read_bytes()).hexdigest()
    lines = ["// Generated by tools/build_indicator_font.py. Do not edit.",
             "// Noto Sans SC, SIL OFL 1.1; see assets/fonts/OFL.txt.",
             f"// Source font SHA256: {digest}", "#pragma once", "#include <stddef.h>",
             "#include <stdint.h>", "namespace ui_font {",
             "struct Character { uint32_t code; uint8_t legacyAdvance; };",
             "struct Glyph { uint32_t offset; uint8_t advance, width, height; int8_t x, y; };",
             "struct Font { uint8_t size, lineHeight; const Glyph *glyphs; const uint8_t *bitmap; };",
             "static const Character kCharacters[] = {"]
    for char in chars:
        lines.append(f"  {{0x{ord(char):04X}, {legacy_advance(legacy_font, char)}}},")
    lines += ["};", "constexpr size_t kCount = sizeof(kCharacters) / sizeof(kCharacters[0]);",
              f"constexpr size_t kFallback = {chars.index('?')};"]
    bitmap_bytes = 0
    for size in PIXEL_SIZES:
        font = ImageFont.truetype(str(font_path), size)
        packed = bytearray()
        lines.append(f"static const Glyph kGlyphs{size}[] = {{")
        for char in chars:
            g = rasterize(font, char)
            lines.append("  {%d, %d, %d, %d, %d, %d}," %
                         (len(packed), g["advance"], g["width"], g["height"], g["x"], g["y"]))
            packed.extend(g["bitmap"])
        lines += ["};", f"static_assert(sizeof(kGlyphs{size}) / sizeof(Glyph) == kCount, \"Incomplete font\");",
                  f"static const uint8_t kBitmap{size}[] = {{"]
        lines.extend("  " + ",".join(f"0x{value:02X}" for value in packed[i:i + 24]) + ","
                     for i in range(0, len(packed), 24))
        lines.append("};")
        bitmap_bytes += len(packed)
    lines += ["static const Font kFonts[] = {"]
    lines += [f"  {{{size}, {math.ceil(18 * size / 16)}, kGlyphs{size}, kBitmap{size}}},"
              for size in PIXEL_SIZES]
    lines += ["};", "constexpr size_t kFontCount = sizeof(kFonts) / sizeof(Font);",
              f"constexpr size_t kBitmapBytes = {bitmap_bytes};", "}", ""]
    output.parent.mkdir(parents=True, exist_ok=True)
    output.write_text("\n".join(lines), encoding="utf-8")
    if preview:
        # A compact proof sheet includes descenders and native large glyphs.
        sample = "gypqj 25:00 RoboPlanter 工位花园"
        sheet = Image.new("RGB", (1100, 320), "#f4f1de")
        y = 8
        for size in PIXEL_SIZES:
            font = ImageFont.truetype(str(font_path), size)
            x = 14
            for char in sample:
                g = rasterize(font, char)
                stride = (g["width"] + 7) // 8
                for row in range(g["height"]):
                    for col in range(g["width"]):
                        if g["bitmap"][row * stride + col // 8] & (0x80 >> (col % 8)):
                            sheet.putpixel((x + g["x"] + col, y + g["y"] + row), (31, 50, 40))
                x += g["advance"]
            y += math.ceil(18 * size / 16) + 7
        preview.parent.mkdir(parents=True, exist_ok=True)
        sheet.save(preview)
    print(f"Font data: {bitmap_bytes:,} bitmap bytes; {len(chars)} characters at each of {PIXEL_SIZES}")
    return len(chars)


def check_coverage(output: Path) -> int:
    """Check every size, including its packed bounds, without requiring the OTF."""
    source = output.read_text(encoding="utf-8")
    match = re.search(r"kCharacters\[\] = \{(.*?)\n\};", source, re.S)
    if not match:
        raise ValueError("Font needs regeneration: native-size character table missing")
    codes = [int(code, 16) for code in re.findall(r"\{0x([0-9A-F]+),", match[1])]
    if codes != sorted(set(codes)):
        raise ValueError("Character table must be unique and sorted")
    missing = [char for char in required_characters() if ord(char) not in codes]
    if missing:
        raise ValueError("Font needs regeneration; missing: " + "".join(missing))
    for size in PIXEL_SIZES:
        entries = re.search(rf"kGlyphs{size}\[\] = \{{(.*?)\n\}};", source, re.S)
        bitmap = re.search(rf"kBitmap{size}\[\] = \{{(.*?)\n\}};", source, re.S)
        if not entries or not bitmap:
            raise ValueError(f"Missing {size} px font")
        rows = re.findall(r"\{(\d+), (\d+), (\d+), (\d+), (-?\d+), (-?\d+)\}", entries[1])
        length = len(re.findall(r"0x[0-9A-F]{2}", bitmap[1]))
        if len(rows) != len(codes):
            raise ValueError(f"Incomplete {size} px font")
        expected_offset = 0
        for row in rows:
            offset, advance, width, height, x, y = map(int, row)
            if (offset != expected_offset or not 0 < advance <= 255
                    or not 0 <= width <= 255 or not 0 <= height <= 255
                    or not -128 <= x <= 127 or not -128 <= y <= 127):
                raise ValueError(f"Invalid {size} px glyph metadata")
            expected_offset += ((width + 7) // 8) * height
        if expected_offset != length:
            raise ValueError(f"Invalid {size} px bitmap bounds")
    return len(codes)


def main() -> None:
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--font", type=Path, default=DEFAULT_FONT)
    parser.add_argument("--output", type=Path, default=DEFAULT_OUTPUT)
    parser.add_argument("--preview", type=Path)
    parser.add_argument("--check", action="store_true", help="Check source character coverage without changing files")
    args = parser.parse_args()
    if args.check:
        try:
            count = check_coverage(args.output)
        except ValueError as error:
            parser.error(str(error))
        print(f"PASS: {count} characters covered at every size {PIXEL_SIZES}")
    else:
        print(f"Generated {generate(args.font, args.output, args.preview)} glyphs: {args.output}")


if __name__ == "__main__":
    main()
